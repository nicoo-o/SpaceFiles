    #[repr(C)]
    struct CompteursMemoire {
        cb: u32,
        fautes_de_page: u32,
        pic_working_set: usize,
        working_set: usize,
        pic_pool_pagine: usize,
        pool_pagine: usize,
        pic_pool_non_pagine: usize,
        pool_non_pagine: usize,
        fichier_pagination: usize,
        pic_fichier_pagination: usize,
        prive: usize,
    }

    #[link(name = "kernel32")]
    extern "system" {
        fn GetCurrentProcess() -> isize;
        fn K32GetProcessMemoryInfo(processus: isize, compteurs: *mut CompteursMemoire, taille: u32) -> i32;
    }

    /// (working set, octets privés) du processus de test, en Mo.
    fn memoire_mo() -> (f64, f64) {
        // SAFETY : structure de taille connue, remplie par l'API pour le processus courant (pseudo-handle valide).
        unsafe {
            let mut c: CompteursMemoire = std::mem::zeroed();
            c.cb = std::mem::size_of::<CompteursMemoire>() as u32;
            K32GetProcessMemoryInfo(GetCurrentProcess(), &mut c, c.cb);
            (c.working_set as f64 / 1048576.0, c.prive as f64 / 1048576.0)
        }
    }

    fn rang(tri: &[f64], p: f64) -> f64 {
        let r = ((p * tri.len() as f64).ceil() as usize).max(1);
        tri[r - 1]
    }

    #[test]
    #[ignore]
    fn bench_requetes_e1() {
        let (ws0, pv0) = memoire_mo();
        println!("MEMOIRE avant chargement : working_set={:.1} Mo prive={:.1} Mo", ws0, pv0);
        let t0 = std::time::Instant::now();
        let idx = charger().expect("index.bin illisible");
        let ms_chargement = t0.elapsed().as_secs_f64() * 1000.0;
        let n = idx.len();
        let (ws1, pv1) = memoire_mo();
        println!("CHARGEMENT index.bin : {:.0} ms, {} elements", ms_chargement, n);
        println!("MEMOIRE apres charger() : working_set={:.1} Mo prive={:.1} Mo (delta ws={:.1} prive={:.1})", ws1, pv1, ws1 - ws0, pv1 - pv0);
        installer(idx);
        for motif in ["a", "windows", "ext:pdf", "readme.md", "note", "zzzqxj", "ext:dll taille>1000000"] {
            let t = std::time::Instant::now();
            let n1 = chercher(motif, MAX_RESULTATS).map(|v| v.len()).unwrap_or(0);
            let premier = t.elapsed().as_secs_f64() * 1000.0;
            let mut ms = Vec::new();
            let mut nr = n1;
            for _ in 0..20 {
                let t = std::time::Instant::now();
                nr = chercher(motif, MAX_RESULTATS).map(|v| v.len()).unwrap_or(0);
                ms.push(t.elapsed().as_secs_f64() * 1000.0);
            }
            ms.sort_by(|a, b| a.partial_cmp(b).unwrap());
            println!(
                "REQUETE {:<24} resultats={:<5} premier_appel={:.1} | essais=20 p50={:.1} p95={:.1} max={:.1} ms",
                motif, nr, premier, rang(&ms, 0.5), rang(&ms, 0.95), ms[ms.len() - 1]
            );
        }
        let (ws2, pv2) = memoire_mo();
        println!("MEMOIRE apres requetes : working_set={:.1} Mo prive={:.1} Mo", ws2, pv2);
    }
